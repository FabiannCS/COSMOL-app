<script>
  // Simple interactive focus traversal for OTP fields
  const otpInputs = document.querySelectorAll('.otp-box');
  otpInputs.forEach((input, idx) => {
    input.addEventListener('input', (e) => {
      if (e.target.value.length === 1 && idx < otpInputs.length - 1) {
        otpInputs[idx + 1].focus();
      }
    });
    input.addEventListener('keydown', (e) => {
      if (e.key === 'Backspace' && !e.target.value && idx > 0) {
        otpInputs[idx - 1].focus();
      }
    });
  });

  // Countdown timer simulation
  let secondsRemaining = 90;
  const timerText = document.getElementById('timer-text');
  const btnResend = document.getElementById('btn-resend');

  const interval = setInterval(() => {
    secondsRemaining--;
    if (secondsRemaining <= 0) {
      clearInterval(interval);
      timerText.innerHTML = '¿No recibiste el código?';
      btnResend.classList.remove('opacity-60', 'pointer-events-none', 'text-outline');
      btnResend.classList.add('text-primary');
      btnResend.removeAttribute('disabled');
    } else {
      const mins = String(Math.floor(secondsRemaining / 60)).padStart(2, '0');
      const secs = String(secondsRemaining % 60).padStart(2, '0');
      timerText.innerHTML = `Reenviar código en <strong class="text-on-surface">${mins}:${secs} min</strong>`;
    }
  }, 1000);
</script></main>
